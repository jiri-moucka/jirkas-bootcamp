# basic operators
- d : directory

- -z : zero length

# Relation operators
- eq
- lt: less than
- le: less than or equal
- gt: greater than
- ge: greater than or equal
- ne: not equal
- -n : not zero length
- -z : zero length

# string operators
- == : if the value of two strings are equal
- != : if the value of two strings are not equal
- -z : if the string is empty
- -n : if the string is not empty
- str : if the string contains the specified substring
